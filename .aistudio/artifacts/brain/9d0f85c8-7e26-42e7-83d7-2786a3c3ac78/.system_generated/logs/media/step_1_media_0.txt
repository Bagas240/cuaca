    (function initWeatherPOV() {
      try {

            isStreetLevel = true;
            if (env) env.style.transform = "rotateX(24deg) translateY(110px) scale(1.15)";

            if (clouds) clouds.style.transform = "translateY(55px) scale(1.15)";

          } else {
            // Atmosphere POV with active rainstorm
            isStreetLevel = false;
            if (env) env.style.transform = "rotateX(12deg) translateY(15px) scale(1.02)";
            if (rainLayer) rainLayer.style.opacity = "0.9";
            if (streetGeom) streetGeom.style.opacity = "0.45";

          }
        }

        // Initialize with default atmospheric rainstorm POV
        setPOVState('atmosphere-rain');

        if (cameraBtn) {
          cameraBtn.addEventListener('click', () => {
            if (isStreetLevel) {
              setPOVState('atmosphere-rain');
            } else {
              setPOVState('street-rain');
            }
          });
        }

        if (stage) {
          stage.addEventListener('click', (e) => {
            if (e.target === stage || e.target === skyBackdrop) {
              if (isStreetLevel) {
                setPOVState('atmosphere-rain');
              } else {
                setPOVState('street-rain');
              }
            }
          });

        function toggleModal(show) {
          if (!modal || !sheet) return;
          if (show) {
            modal.classList.remove('opacity-0', 'pointer-events-none');
            modal.classList.add('opacity-100', 'pointer-events-auto');
            sheet.classList.remove('translate-y-full');
            sheet.classList.add('translate-y-0');
          } else {
            modal.classList.add('opacity-0', 'pointer-events-none');
            modal.classList.remove('opacity-100', 'pointer-events-auto');
            sheet.classList.add('translate-y-full');
            sheet.classList.remove('translate-y-0');
          }
        }

        if (openModalBtn) openModalBtn.addEventListener('click', () => toggleModal(true));
        if (openSearchBtn) openSearchBtn.addEventListener('click', () => toggleModal(true));
        if (closeModalBtn) closeModalBtn.addEventListener('click', () => toggleModal(false));
        if (modal) {
          modal.addEventListener('click', (e) => {
            if (e.target === modal) toggleModal(false);
          });
        }

      } catch (err) {
        console.error("Weather POV init notice:", err);
      }
    })();

    (function initSmogCamera() {
      try {
        const stage = document.getElementById('smogPOV');
        const grid = document.getElementById('smogCityGrid');
        let currentPitch = 28;
        let currentYaw = -6;
        let isDragging = false;
        let startX = 0;
        let startY = 0;

        function updateTransform() {
          if (!grid) return;
          grid.style.transform = `rotateX(${currentPitch}deg) rotateY(${currentYaw}deg) translateZ(-20px)`;
        }

        if (stage) {
          stage.addEventListener('mousedown', (e) => {
            isDragging = true;
            startX = e.clientX;
            startY = e.clientY;
          });
          window.addEventListener('mousemove', (e) => {
            if (!isDragging) return;
            const deltaX = e.clientX - startX;
            const deltaY = e.clientY - startY;
            startX = e.clientX;
            startY = e.clientY;
            currentYaw = Math.max(-25, Math.min(25, currentYaw + deltaX * 0.15));
            currentPitch = Math.max(10, Math.min(50, currentPitch - deltaY * 0.15));
            updateTransform();
          });
          window.addEventListener('mouseup', () => { isDragging = false; });
        }
      } catch (e) {
        console.error("AQI 3D Camera notice:", e);
      }
    })();

    (function initTectonicCamera() {
      try {
        const stage = document.getElementById('tectonicPOV');
        const mesh = document.getElementById('tectonicMesh');
        let currentPitch = 32;
        let currentYaw = -8;
        let isDragging = false;
        let startX = 0;
        let startY = 0;

        function updateTransform() {
          if (!mesh) return;
          mesh.style.transform = `rotateX(${currentPitch}deg) rotateY(${currentYaw}deg) translateY(-20px)`;
        }

        if (stage) {
          stage.addEventListener('mousedown', (e) => {
            isDragging = true;
            startX = e.clientX;
            startY = e.clientY;
          });
          window.addEventListener('mousemove', (e) => {
            if (!isDragging) return;
            const deltaX = e.clientX - startX;
            const deltaY = e.clientY - startY;
            startX = e.clientX;
            startY = e.clientY;
            currentYaw = Math.max(-25, Math.min(25, currentYaw + deltaX * 0.15));
            currentPitch = Math.max(15, Math.min(55, currentPitch - deltaY * 0.15));
            updateTransform();
          });
          window.addEventListener('mouseup', () => { isDragging = false; });
        }
      } catch (e) {
        console.error("Disaster monitor POV notice:", e);
      }
    })();

    (function initRadarCamera() {
      try {
        const stage = document.getElementById('radarPOV');
        const mesh = document.getElementById('globeMesh');
        let currentPitch = 24;
        let currentYaw = -10;
        let isDragging = false;
        let startX = 0;
        let startY = 0;

        function updateTransform() {
          if (!mesh) return;
          mesh.style.transform = `rotateX(${currentPitch}deg) rotateY(${currentYaw}deg) scale(1.05)`;
        }

        if (stage) {
          stage.addEventListener('mousedown', (e) => {
            isDragging = true;
            startX = e.clientX;
            startY = e.clientY;
          });
          window.addEventListener('mousemove', (e) => {
            if (!isDragging) return;
            const deltaX = e.clientX - startX;
            const deltaY = e.clientY - startY;
            startX = e.clientX;
            startY = e.clientY;
            currentYaw = Math.max(-25, Math.min(25, currentYaw + deltaX * 0.15));
            currentPitch = Math.max(10, Math.min(50, currentPitch - deltaY * 0.15));
            updateTransform();
          });
          window.addEventListener('mouseup', () => { isDragging = false; });
        }
      } catch (e) {
        console.error("Radar POV notice:", e);
      }
    })();
  </script>
